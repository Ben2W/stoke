## Install

```sh
pnpm add @rigkit/provider-cmux
```

## Configure

```ts rig.config.ts
import { defineProject } from "@rigkit/sdk";
import { cmux } from "@rigkit/provider-cmux";

export default defineProject({
  providers: {
    sessions: cmux({}),
  },
});
```
